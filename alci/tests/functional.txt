## Alci functional test suite. One command per line, max ~60 chars
## (ELKS tty input line is 80 chars). Output must be deterministic.
## Run with alci/tests/qtest.py on QEMU; compare against upstream.
export PATH=/bin
## --- shell
(cd /bin; pwd)
echo X$(pwd)Y$(ls /bin | wc -l)Z
for i in 1 2 3; do echo n$i; done
sh -c 'for i in a b; do echo in-$i; done'
false || echo or-ok; true && echo and-ok
case abc in a*) echo case-ok;; *) echo bad;; esac
x=1; while test $x -lt 4; do echo w$x; x=$(expr $x + 1); done
## --- fork and processes
sleep 1 & sleep 1 & sleep 1 & wait; echo bg3-ok
for i in 1 2 3 4 5 6 7 8; do sh -c 'echo p' & done; wait
sh -c 'sh -c "sh -c \"sh -c pwd\""'
cd /bin; ls | head -20 | xargs ls -d | wc -l; cd /
## --- files and pipes
dd if=/bin/sh of=/tmp/a bs=512 2>/dev/null
cmp /bin/sh /tmp/a && echo dd-ok
cat /bin/sh | cat | cat > /tmp/p; cmp /bin/sh /tmp/p
cp /bin/vi /tmp/b; cmp /bin/vi /tmp/b && echo cp-ok
cd /tmp; tar cf t.tar a b; mkdir x; cd x; tar xf ../t.tar
cmp a /bin/sh && cmp b /bin/vi && echo tar-ok; cd /
compress -c /bin/sort > /tmp/c.Z; ls -l /tmp/c.Z | wc -l
compress -dc /tmp/c.Z | md5sum; md5sum /bin/sort
sort -r /etc/passwd | head -2
ls /bin | sort | uniq | tail -3
yes abc | head -500 | sort | uniq -c
rm -rf /tmp/p /tmp/a /tmp/b /tmp/x /tmp/t.tar /tmp/c.Z; ls /tmp
mkdir /tmp/d; touch /tmp/d/f1 /tmp/d/f2; ls /tmp/d
rm -r /tmp/d; ls /tmp; sync; echo sync-ok
## --- libc
test_libc 2>&1 | tail -3
test_float 2>&1 | tail -2
